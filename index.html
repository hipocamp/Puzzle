    <script>

        const board = document.getElementById('puzzle-board');

        const winScreen = document.getElementById('win-screen');

        let pieces = [0, 1, 2, 3, 4, 5, 6, 7, 8]; // 8 és el forat buit

        

        // Crear el tauler inicial

        function createBoard() {

            board.innerHTML = '';

            pieces.forEach((pos, index) => {

                const div = document.createElement('div');

                div.className = 'piece';

                if (pos === 8) {

                    div.classList.add('empty');

                } else {

                    // Càlcul de la part de la imatge que ha de mostrar cada peça

                    const row = Math.floor(pos / 3);

                    const col = pos % 3;

                    div.style.backgroundPosition = `${col * 50}% ${row * 50}%`;

                }

                

                div.onclick = () => movePiece(index);

                board.appendChild(div);

            });

        }


        function movePiece(index) {

            const emptyIndex = pieces.indexOf(8);

            const validMoves = getValidMoves(emptyIndex);


            if (validMoves.includes(index)) {

                // Intercanviar posicions

                [pieces[index], pieces[emptyIndex]] = [pieces[emptyIndex], pieces[index]];

                createBoard();

                checkWin();

            }

        }


        function getValidMoves(emptyIdx) {

            const moves = [];

            const row = Math.floor(emptyIdx / 3);

            const col = emptyIdx % 3;


            if (row > 0) moves.push(emptyIdx - 3); // A dalt

            if (row < 2) moves.push(emptyIdx + 3); // A baix

            if (col > 0) moves.push(emptyIdx - 1); // Esquerra

            if (col < 2) moves.push(emptyIdx + 1); // Dreta

            

            return moves;

        }


        function shuffleGame() {

            winScreen.style.display = 'none';

            // Perquè sigui resoluble, fem moviments aleatoris reals des de la posició resolta

            let iterations = 0;

            let currentEmpty = pieces.indexOf(8);

            

            const interval = setInterval(() => {

                const moves = getValidMoves(pieces.indexOf(8));

                const randomMove = moves[Math.floor(Math.random() * moves.length)];

                

                const emptyIdx = pieces.indexOf(8);

                [pieces[randomMove], pieces[emptyIdx]] = [pieces[emptyIdx], pieces[randomMove]];

                

                createBoard();

                iterations++;

                if (iterations > 30) clearInterval(interval);

            }, 50);

        }


        function checkWin() {

            const isWin = pieces.every((val, i) => val === i);

            if (isWin) {

                setTimeout(() => {

                    winScreen.style.display = 'flex';

                }, 500);

            }

        }


        // Iniciar

        createBoard();

    </script>
